GoIT RDB HW-03 - SQL Queries
========================================

Task 1: Select all columns from products table and name, phone from shippers table
----------------------------------------------------------------------------------
-- Select all columns from products table using wildcard
SELECT * FROM products;

-- Select only name and phone columns from shippers table
SELECT name, phone FROM shippers;


Task 2: Find average, maximum, and minimum price values from products table
---------------------------------------------------------------------------
SELECT 
    AVG(price) AS average_price,
    MAX(price) AS max_price,
    MIN(price) AS min_price
FROM products;


Task 3: Select unique category_id and price from products, ordered by price descending, limited to 10 rows
----------------------------------------------------------------------------------------------------
SELECT DISTINCT category_id, price
FROM products
ORDER BY price DESC
LIMIT 10;


Task 4: Count products with price between 20 and 100
---------------------------------------------------
SELECT COUNT(*) AS product_count
FROM products
WHERE price BETWEEN 20 AND 100;


Task 5: Count products and average price grouped by supplier_id
-----------------------------------------------------------------
SELECT 
    supplier_id,
    COUNT(*) AS product_count,
    AVG(price) AS average_price
FROM products
GROUP BY supplier_id;
